export function showArchiveButton(document) {
  return (document.status === 'completed' || document.status === 'cancelled')
    && !document.locked;
}
export function documentActions(document) {
  const actions = ['open'];
  if ((document.status === 'completed' || document.status === 'cancelled')
      && !document.locked) {
    actions.push('archive');
  }
  return actions;
}

export function canPin(document) {
  return (document.status === 'completed' || document.status === 'cancelled')
    && !document.locked;
}
export function archiveIds(documents) {
  return documents
    .filter(document => (document.status === 'completed' || document.status === 'cancelled')
      && !document.locked)
    .map(document => document.id);
}

const cases = [
  ['draft', false, false],
  ['draft', true, false],

  test(`${status}, locked=${locked}`, () => {
    const document = Object.freeze({id: 'd1', status, locked});

    assert.deepEqual(documentActions(document), allowed ? ['open', 'archive'] : ['open']);
    assert.deepEqual(archiveIds(Object.freeze([document])), allowed ? ['d1'] : []);
    assert.equal(canPin(document), allowed);
  });
}
